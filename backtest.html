function backtest(candles, asset, params, costsPct, maxBars, strategy="reversal") { const closes=candles.map(c=>c.close); const bb=bollinger(closes,params.bb,params.dev); const rs=rsi(closes,params.rsi); const at=atr(candles,14); const ax=adx(candles,14); function ema(values, period) { const out=Array(values.length).fill(null); const k=2/(period+1); let value=null; for(let i=0;i=period-1)out[i]=value; } return out; } const e9=ema(closes,9); const e21=ema(closes,21); const trades=[]; let pos=null; const minBars=Math.max(params.bb,params.rsi,28)+2; for(let i=minBars;i=pos.stop){ exit=pos.stop;reason="Стоп"; } else if(pos.side==="LONG" && c.high>=pos.target){ exit=pos.target;reason="Цель"; } else if(pos.side==="SHORT" && c.low<=pos.target){ exit=pos.target;reason="Цель"; } else if(held>=maxBars){ exit=c.close;reason="Лимит времени"; } if(exit!==null){ const raw=pos.side==="LONG" ?(exit-pos.entry)/pos.risk :(pos.entry-exit)/pos.risk; const costR=(pos.entry*(costsPct/100))/pos.risk; trades.push({ asset,time:c.time,side:pos.side, entry:pos.entry,exit,risk:pos.risk, rawR:raw,costR,R:raw-costR,reason,held, strategy }); pos=null; } continue; } if(!at[i] || !ax[i] || !e9[i] || !e21[i])continue; let side=null; let target=null; if(strategy==="trend"){ // По тренду: EMA 9/21, направление свечи и ADX от 18. if(ax[i]>=18 && e9[i]>e21[i] && c.close>e9[i] && c.close>c.open){ side="LONG"; } else if(ax[i]>=18 && e9[i]params.adx)continue; if(prev.close=bb.lower[i] && rs[i]<=params.low){ side="LONG"; } else if(prev.close>bb.upper[i-1] && c.close<=bb.upper[i] && rs[i]>=params.high){ side="SHORT"; } } if(!side)continue; const entry=c.close; const risk=at[i]*params.atr; if(!Number.isFinite(risk) || risk<=0)continue; if(strategy==="trend"){ // Цель трендовой сделки — 1.5 первоначального риска. target=side==="LONG" ?entry+risk*1.5 :entry-risk*1.5; } else { target=bb.mid[i]; const reward=side==="LONG"?target-entry:entry-target; if(reward<=0 || reward/risk0){wins++;grossWin+=t.R;} else if(t.R<0){grossLoss+=Math.abs(t.R);} eq+=t.R; peak=Math.max(peak,eq); maxDD=Math.max(maxDD,peak-eq); } return { asset,strategy,trades,count:trades.length, winRate:trades.length?wins/trades.length*100:0, total, profitFactor:grossLoss?grossWin/grossLoss: (grossWin?Infinity:0), maxDD,wins,losses:trades.length-wins,params }; }
